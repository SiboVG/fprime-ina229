cmake_minimum_required(VERSION 3.16)
project(Ina229HostTests LANGUAGES CXX)
enable_testing()
option(INA229_COVERAGE "Enable host coverage" OFF)
add_library(ina229_core ../Ina229/Device/Device.cpp)
target_include_directories(ina229_core PUBLIC ..)
target_compile_features(ina229_core PUBLIC cxx_std_11)
target_compile_options(ina229_core PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(ina229_tests DeviceTests.cpp)
target_link_libraries(ina229_tests PRIVATE ina229_core)
target_compile_options(ina229_tests PRIVATE -Wall -Wextra -Werror -pedantic)
if(INA229_COVERAGE)
 target_compile_options(ina229_core PRIVATE --coverage)
 target_link_libraries(ina229_tests PRIVATE --coverage)
endif()
add_test(NAME ina229_device COMMAND ina229_tests)

if(UNIX)
 find_package(Threads REQUIRED)
 add_executable(ina229_bridge_tests BridgeTests.cpp ../examples/Ft232h/UnixSpi.cpp)
 target_link_libraries(ina229_bridge_tests PRIVATE ina229_core Threads::Threads)
 target_compile_options(ina229_bridge_tests PRIVATE -Wall -Wextra -Werror -pedantic)
 add_test(NAME ina229_bridge COMMAND ina229_bridge_tests)
endif()
